// ===== CÓDIGO DA PROFESSORA ===== // É pedido na aplicação para lançar a nota do caderno diário no computador do aluno. // Escolhe um código só teu e escreve-o aqui, entre as aspas. var CODIGO_PROFESSORA = 'escolhe-um-codigo'; function doPost(e) { try { var data = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.getActiveSpreadsheet(); var tipo = String(data.tipo || ''); // Nota do caderno diário, lançada pela professora na aplicação (com o código). if (tipo === 'Caderno') { return respostaJson_(guardarCaderno_(ss, data)); } // Sítio onde cada aluno está — uma linha por aluno na aba "Progresso". if (tipo === 'Progresso') { guardarProgresso_(ss, data); return respostaJson_({ status: 'ok' }); } // Saídas (com ou sem reinício), reflexões e projetos vão para "Missão - Registos". if (tipo.indexOf('Saída do separador') === 0 || tipo === 'Reflexão' || tipo === 'Projeto') { var sheet = ss.getSheetByName('Missão - Registos') || ss.insertSheet('Missão - Registos'); if (sheet.getLastRow() === 0) { sheet.appendRow([ 'Data/Hora', 'Nome', 'Turma', 'Nível', 'Tipo', 'Momento', 'Texto da Reflexão', 'Avaliação da reflexão', 'Material necessário' ]); // Menu suspenso (dropdown) na coluna "Avaliação da reflexão" (coluna H), // já preparado para as próximas 500 linhas. var avaliacaoRange = sheet.getRange('H2:H500'); var regra = SpreadsheetApp.newDataValidation() .requireValueInList(['Respondeu bem', 'Respondeu parcialmente', 'Não respondeu'], true) .setAllowInvalid(false) .build(); avaliacaoRange.setDataValidation(regra); } if (tipo === 'Reflexão' || tipo === 'Projeto') { // Respostas e projeto: atualiza a linha que já existe (não repete linhas). guardarResposta_(sheet, data); } else { sheet.appendRow([ new Date(), data.nome || '', data.turma || '', data.nivel || '', data.tipo || '', data.momento || '', data.texto || '', '', data.material || '' ]); } } else { var sheet2 = ss.getSheetByName('Resultados') || ss.insertSheet('Resultados'); if (sheet2.getLastRow() === 0) { sheet2.appendRow([ 'Data/Hora', 'Nome', 'Turma', 'Número de Aluno', 'Nível', 'Pontuação', '% Respostas Certas', 'Tempo Total (mm:ss)', 'Saídas do Separador', 'Tempo Médio por Item (s)', 'Sinal de Alerta', 'Nota (0-20)', 'Modo' ]); } // "Modo" distingue Missão (treino) de Prova (exame). Quando não vem no pedido // (ex.: submissões antigas da Prova, feitas antes desta coluna existir), // assume-se "Prova" para manter a compatibilidade com o que já tinhas. sheet2.appendRow([ new Date(), data.nome || '', data.turma || '', data.numeroAluno || '', data.nivel || '', data.pontuacao || '', data.percentagem || '', data.tempoTotal || '', data.saidas || '', data.tempoMedioPorItem || '', data.sinalRapido || '', data.nota || '', data.modo || 'Prova' ]); } return respostaJson_({ status: 'ok' }); } catch (err) { return respostaJson_({ status: 'error', message: err.message }); } } function doGet(e) { // A aplicação pede as respostas de um aluno num nível, com a avaliação da professora: // ?acao=respostas&nome=...&turma=...&nivel=... if (e && e.parameter && e.parameter.acao === 'respostas') { try { var ssR = SpreadsheetApp.getActiveSpreadsheet(); var reg = ssR.getSheetByName('Missão - Registos'); var out = { ok: true, reflexoes: {}, projeto: null }; if (reg && reg.getLastRow() > 1) { var n = normalizarAluno_(e.parameter.nome); var t = normalizarAluno_(e.parameter.turma).replace(/ /g, ''); var nivel = String(e.parameter.nivel || ''); var v = reg.getRange(2, 1, reg.getLastRow() - 1, 9).getValues(); v.forEach(function (r) { if (String(r[3]) !== nivel) return; if (normalizarAluno_(r[1]) !== n || normalizarAluno_(r[2]).replace(/ /g, '') !== t) return; var tipoR = String(r[4]); if (tipoR === 'Reflexão') { var m = String(r[5]).trim(); var nova = { texto: String(r[6]), avaliacao: String(r[7]).trim() }; // Se houver linhas repetidas, fica a avaliada; senão, a mais recente. if (!out.reflexoes[m] || !out.reflexoes[m].avaliacao || nova.avaliacao) out.reflexoes[m] = nova; } else if (tipoR === 'Projeto') { out.projeto = { texto: String(r[6]), material: String(r[8]) }; } }); } // Nota e comentário do projeto (escritos pela professora na aba "Acompanhamento"). try { out.avaliacaoProjeto = avaliacaoProjeto_(ssR, e.parameter.nome, Number(e.parameter.nivel)); } catch (errProjeto) { out.avaliacaoProjeto = null; } // Nota do caderno diário deste aluno neste nível (aba "Caderno diário"). try { out.caderno = lerNotaCaderno_(ssR, e.parameter.nome, e.parameter.nivel); } catch (errCaderno) { out.caderno = ''; } return respostaJson_(out); } catch (err) { return respostaJson_({ ok: false, erro: err.message }); } } // A aplicação pergunta onde o aluno ficou: ?acao=progresso&chave=... if (e && e.parameter && e.parameter.acao === 'progresso') { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName('Progresso'); var linha = sheet ? procurarLinhaAluno_(sheet, String(e.parameter.chave || '')) : -1; if (linha === -1) { return respostaJson_({ encontrado: false }); } var v = sheet.getRange(linha, 1, 1, 13).getValues()[0]; return respostaJson_({ encontrado: true, nome: String(v[0]), turma: String(v[1]), nivelAtual: String(v[3]), passo: String(v[4]), onde: String(v[5]), concluidos: String(v[6]), pontos: String(v[7]), proposta: String(v[8]), material: String(v[9]), inicioProjeto: String(v[10]), // Dia do último registo (fuso horário da folha), para a aplicação saber // se o aluno voltou no mesmo dia (conta como saída) ou noutro dia. dia: (v[11] instanceof Date) ? Utilities.formatDate(v[11], ss.getSpreadsheetTimeZone(), 'yyyy-MM-dd') : '' }); } catch (err) { return respostaJson_({ encontrado: false, erro: err.message }); } } return ContentService.createTextOutput('O script está a funcionar. Usa POST para enviar dados.'); } // ===== Funções de apoio ===== function respostaJson_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } function folhaProgresso_(ss) { var sheet = ss.getSheetByName('Progresso'); if (!sheet) { sheet = ss.insertSheet('Progresso'); sheet.appendRow([ 'Nome', 'Turma', 'Chave', 'Nível em curso', 'Passo (nº)', 'Onde está', 'Níveis concluídos', 'Pontos', 'Projeto (rascunho)', 'Material (rascunho)', 'Início do projeto', 'Última atualização', 'Sessão' ]); // Colunas A a K e M em texto simples, para o Sheets não transformar // "1, 2" num número decimal nem um texto começado por "=" numa fórmula. sheet.getRange('A:K').setNumberFormat('@'); sheet.getRange('M:M').setNumberFormat('@'); sheet.setFrozenRows(1); } return sheet; } function procurarLinhaAluno_(sheet, chave) { var ultima = sheet.getLastRow(); if (ultima < 2 || !chave) return -1; var chaves = sheet.getRange(2, 3, ultima - 1, 1).getValues(); for (var i = 0; i < chaves.length; i++) { if (String(chaves[i][0]) === chave) return i + 2; } return -1; } function guardarProgresso_(ss, data) { // Bloqueio para dois pedidos ao mesmo tempo não criarem linhas repetidas. var lock = LockService.getScriptLock(); lock.waitLock(20000); try { var sheet = folhaProgresso_(ss); var chave = String(data.chave || ''); if (!chave) return; var valores = [ String(data.nome || ''), String(data.turma || ''), chave, String(data.nivelAtual || ''), (data.passo === undefined || data.passo === null) ? '' : String(data.passo), String(data.onde || ''), String(data.concluidos || ''), String(data.pontos || 0), String(data.proposta || ''), String(data.material || ''), String(data.inicioProjeto || ''), new Date(), String(data.sessao || '') + ':' + String(data.seq || 0) ]; var linha = procurarLinhaAluno_(sheet, chave); if (linha === -1) { sheet.appendRow(valores); return; } // Ignora um pedido atrasado da MESMA sessão (não recua o aluno). // Não usa o relógio dos computadores, que pode estar desacertado. var guardado = String(sheet.getRange(linha, 13).getValue()).split(':'); if (guardado[0] && guardado[0] === String(data.sessao || '') && (Number(data.seq) || 0) < (Number(guardado[1]) || 0)) return; sheet.getRange(linha, 1, 1, 13).setValues([valores]); } finally { lock.releaseLock(); } } // Compara nomes e turmas sem maiúsculas, acentos nem espaços a mais. function normalizarAluno_(s) { var mapa = { 'á':'a','à':'a','â':'a','ã':'a','ä':'a','é':'e','è':'e','ê':'e','ë':'e','í':'i','ì':'i','î':'i','ï':'i', 'ó':'o','ò':'o','ô':'o','õ':'o','ö':'o','ú':'u','ù':'u','û':'u','ü':'u','ç':'c','ñ':'n' }; return String(s || '').replace(/[\u200B-\u200D\u2060\uFEFF\u00AD]/g, '').toLowerCase() .replace(/[áàâãäéèêëíìîïóòôõöúùûüçñ]/g, function (ch) { return mapa[ch]; }) .replace(/[^a-z0-9]+/g, ' ').trim(); } // Procura, de baixo para cima, a linha da resposta deste aluno (Reflexão: o mesmo momento; // Projeto: o projeto desse nível). Devolve -1 se ainda não existir. function procurarResposta_(sheet, data) { var ultima = sheet.getLastRow(); if (ultima < 2) return -1; var v = sheet.getRange(2, 1, ultima - 1, 6).getValues(); var n = normalizarAluno_(data.nome); var t = normalizarAluno_(data.turma).replace(/ /g, ''); for (var i = v.length - 1; i >= 0; i--) { var r = v[i]; if (String(r[4]) !== String(data.tipo)) continue; if (String(r[3]) !== String(data.nivel)) continue; if (data.tipo === 'Reflexão' && String(r[5]).trim() !== String(data.momento || '').trim()) continue; if (normalizarAluno_(r[1]) !== n || normalizarAluno_(r[2]).replace(/ /g, '') !== t) continue; return i + 2; } return -1; } function guardarResposta_(sheet, data) { var lock = LockService.getScriptLock(); lock.waitLock(20000); try { var linha = procurarResposta_(sheet, data); if (linha === -1) { sheet.appendRow([ new Date(), data.nome || '', data.turma || '', data.nivel || '', data.tipo || '', data.momento || '', data.texto || '', '', data.material || '' ]); return; } // Resposta já avaliada pela professora: fica como está. if (data.tipo === 'Reflexão' && String(sheet.getRange(linha, 8).getValue()).trim() !== '') return; sheet.getRange(linha, 1).setValue(new Date()); sheet.getRange(linha, 7).setValue(data.texto || ''); if (data.tipo === 'Projeto') sheet.getRange(linha, 9).setValue(data.material || ''); } finally { lock.releaseLock(); } } // ===== CADERNO DIÁRIO ===== // Uma linha por aluno e por nível na aba "Caderno diário". Podes também alterar a nota // diretamente nessa aba. function folhaCaderno_(ss) { var sheet = ss.getSheetByName('Caderno diário'); if (!sheet) { sheet = ss.insertSheet('Caderno diário'); sheet.appendRow(['Nome', 'Turma', 'Nível', 'Nota (0-20)', 'Última alteração']); sheet.setFrozenRows(1); } return sheet; } function procurarCaderno_(sheet, nome, nivel) { var ultima = sheet.getLastRow(); if (ultima < 2) return -1; var v = sheet.getRange(2, 1, ultima - 1, 3).getValues(); var n = normalizarAluno_(nome); for (var i = v.length - 1; i >= 0; i--) { if (String(v[i][2]) === String(nivel) && normalizarAluno_(v[i][0]) === n) return i + 2; } return -1; } function guardarCaderno_(ss, data) { if (String(data.codigo || '') !== String(CODIGO_PROFESSORA)) return { status: 'erro', motivo: 'codigo' }; var nota = parseFloat(String(data.nota).replace(',', '.')); if (isNaN(nota) || nota < 0 || nota > 20) return { status: 'erro', motivo: 'nota' }; var lock = LockService.getScriptLock(); lock.waitLock(20000); try { var sheet = folhaCaderno_(ss); var linha = procurarCaderno_(sheet, data.nome, data.nivel); var valores = [String(data.nome || ''), String(data.turma || ''), Number(data.nivel) || '', nota, new Date()]; if (linha === -1) sheet.appendRow(valores); else sheet.getRange(linha, 1, 1, 5).setValues([valores]); } finally { lock.releaseLock(); } return { status: 'ok', nota: nota }; } function lerNotaCaderno_(ss, nome, nivel) { var sheet = ss.getSheetByName('Caderno diário'); if (!sheet) return ''; var linha = procurarCaderno_(sheet, nome, nivel); if (linha === -1) return ''; var v = sheet.getRange(linha, 4).getValue(); return (v === '' || v === null) ? '' : String(v); } // ===== AVALIAÇÃO DO PROJETO (para o aluno ver na aplicação) ===== // Lê, na aba "Acompanhamento", a nota e o comentário do projeto deste aluno neste nível. // Usa as funções do ficheiro "acompanhamento" (estão no mesmo projeto do Apps Script). function avaliacaoProjeto_(ss, nome, nivel) { if (typeof lerColunasAcompanhamento_ !== 'function') return null; var sh = ss.getSheetByName('Acompanhamento'); if (!sh) return null; var tudo = lerColunasAcompanhamento_(sh); var aluno = null; if (ss.getSheetByName('Alunos') && typeof identificarAluno_ === 'function') { aluno = identificarAluno_(nome, lerAlunos_(ss)); } if (!aluno) { var n = normalizarAluno_(nome); for (var k in tudo) if (normalizarAluno_(k) === n) { aluno = k; break; } } if (!aluno || !tudo[aluno] || !tudo[aluno][nivel]) return null; var o = tudo[aluno][nivel]; var nota = (o.projeto !== undefined && o.projeto !== null) ? String(o.projeto) : ''; var comentario = o.comentario ? String(o.comentario) : ''; if (!nota && !comentario) return null; return { nota: nota, comentario: comentario }; }
